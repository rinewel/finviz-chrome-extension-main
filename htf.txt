// 把标题标签"尺寸"枚举转成 Pine 内置的 size.* 常量
toLabelSize(LabelSize labelSize) =>
    switch labelSize
        LabelSize.tiny   => size.tiny
        LabelSize.small  => size.small
        LabelSize.normal => size.normal
        LabelSize.large  => size.large
        LabelSize.huge   => size.huge

// 把标题标签"样式"枚举转成 Pine 内置的 label.style_* 常量
toLabelStyle(LabelStyle labelStyle) =>
    switch labelStyle
        LabelStyle.down   => label.style_label_down
        LabelStyle.up     => label.style_label_up
        LabelStyle.left   => label.style_label_left
        LabelStyle.right  => label.style_label_right
        LabelStyle.center => label.style_label_center

// 把周期字符串转成可读文字，比如 "240" → "4小时"，"D" → "1日"，"15" → "15分钟"，"W" → "1周"，"30S" → "30秒"
// 直接解析字符串，避免月线（1M = 2628003 秒）、周线、秒级周期按分钟换算出错
timeframeText(string tf) =>
    string tfStr   = tf == "" ? timeframe.period : tf
    string unit    = str.substring(tfStr, str.length(tfStr) - 1)
    bool   hasUnit = unit == "D" or unit == "W" or unit == "M" or unit == "S"
    string numText = hasUnit ? str.substring(tfStr, 0, str.length(tfStr) - 1) : tfStr
    int    mult    = numText == "" ? 1 : int(str.tonumber(numText))
    switch
        unit == "D"    => str.tostring(mult) + "日"
        unit == "W"    => str.tostring(mult) + "周"
        unit == "M"    => str.tostring(mult) + "月"
        unit == "S"    => str.tostring(mult) + "秒"
        mult % 60 == 0 => str.tostring(int(mult / 60)) + "小时"
        => str.tostring(mult) + "分钟"

// 按选定类型计算移动平均
movingAverage(float src, simple int length, simple MaType maType) =>
    switch maType
        MaType.ema  => ta.ema(src, length)
        MaType.sma  => ta.sma(src, length)
        MaType.wma  => ta.wma(src, length)
        MaType.rma  => ta.rma(src, length)
        MaType.vwma => ta.vwma(src, length)

// 判断取回的K线数据是否无效：实时K线的开高低收任意一个为 na 即视为取数失败
isDataInvalid(array<float> ohlcData) =>
    na(ohlcData.get(0)) or na(ohlcData.get(1)) or na(ohlcData.get(2)) or na(ohlcData.get(3))

// 在被请求的周期上计算一条云图的快慢均线，返回最近10根的值（共20个：0~9快线，10~19慢线）
// 不显示的云图不计算均线，直接返回全 na（isShown 来自输入，整段历史中不会变化，所以条件计算是安全的）
cloudSeries(simple bool isShown, simple int fastLen, simple int slowLen) =>
    array<float> values = array.new<float>(CLOUD_SIZE, na)
    if isShown
        float fastMa = movingAverage(maSourceInput, fastLen, maTypeInput)
        float slowMa = movingAverage(maSourceInput, slowLen, maTypeInput)
        values := array.from(
             fastMa, fastMa[1], fastMa[2], fastMa[3], fastMa[4], fastMa[5], fastMa[6], fastMa[7], fastMa[8], fastMa[9],
             slowMa, slowMa[1], slowMa[2], slowMa[3], slowMa[4], slowMa[5], slowMa[6], slowMa[7], slowMa[8], slowMa[9])
    values

// 在被请求的周期上一次算完一组所需的全部数据（布局见常量 GROUP_DATA_SIZE）：
// 最近10根K线的开高低收（偏移量*4 + 0/1/2/3 = 开/高/低/收；偏移量0 = 实时K线）+ 5条云图的快慢均线
// 均线在这里计算、整体作为 request.security 的表达式，确保均线是在高周期上计算的
htfGroupValues(simple bool show1, simple int fast1, simple int slow1, simple bool show2, simple int fast2, simple int slow2, simple bool show3, simple int fast3, simple int slow3, simple bool show4, simple int fast4, simple int slow4, simple bool show5, simple int fast5, simple int slow5) =>

         open,    high,    low,    close,
         open[1], high[1], low[1], close[1],
         open[2], high[2], low[2], close[2],
         open[3], high[3], low[3], close[3],
         open[4], high[4], low[4], close[4],
         open[5], high[5], low[5], close[5],
         open[6], high[6], low[6], close[6],
         open[7], high[7], low[7], close[7],
         open[8], high[8], low[8], close[8],
         open[9], high[9], low[9], close[9])

    data.concat(cloudSeries(show2, fast2, slow2))

    data

// 计算一组内每根K线的左边界/右边界/中点横坐标（K线与云图共用，保证对齐）
// liveAnchor：实时K线左边界；历史K线从锚点往左排开，并在自己的"宽度+间距"槽位里居中
// 超出右侧绘制上限时的左移在主流程里对三组整体进行，避免各组单独左移后互相重叠
calcGroupPositions(int count, int liveAnchor, int candleWidth, int candleGap) =>
    int halfGap = int(candleGap / 2)
    int halfWidth = int(candleWidth / 2)
    array<int> startXs = array.new<int>(count, 0)
    array<int> endXs   = array.new<int>(count, 0)
    array<int> midXs   = array.new<int>(count, 0)
    for p = 0 to count - 1
        int distFromLive = count - 1 - p
        int startX = distFromLive == 0 ? liveAnchor : liveAnchor - distFromLive * (candleWidth + candleGap) + halfGap
        startXs.set(p, startX)
        endXs.set(p, startX + candleWidth)
        midXs.set(p, startX + halfWidth)
    [startXs, endXs, midXs]

// ─────────────────────────────────────────────
// 绘图对象池
// 每组的图形只创建一次，之后每次刷新只用 set_* 更新位置、文字和颜色，用不到的设为透明
// 避免每个报价都删除并重建几百个线、填充、框和标签
// ─────────────────────────────────────────────
const int SEGMENT_COUNT    = HTF_BAR_COUNT - 1               // 每条云图最多的线段数（相邻两根K线之间一段）
const int CLOUD_LINE_COUNT = CLOUD_COUNT * SEGMENT_COUNT     // 每组云图线段总数

type GroupDrawings
    array<line>     wicks           // 影线，下标 = 从左到右第几根K线
    array<box>      bodies          // 实体，下标同上
    array<line>     dashLines       // 实时K线的开/高/低/收虚线连接线
    array<label>    priceLabels     // 实时K线的开/高/低/收价格标签
    label           titleLabel      // 标题标签（品种代码 + 周期）
    label           warningLabel    // 诊断提示标签
    array<line>     cloudFastLines  // 云图快线段，下标 = 云图序号 * SEGMENT_COUNT + 段序号
    array<line>     cloudSlowLines  // 云图慢线段，下标同上
    array<linefill> cloudFills      // 快慢线段之间的填充，下标同上

newHiddenLine(int width) =>
    line.new(bar_index, close, bar_index, close, color = TRANSPARENT, width = width)

newHiddenLabel(string style, string textSize) =>
    label.new(bar_index, close, "", color = TRANSPARENT, textcolor = TRANSPARENT, style = style, size = textSize)

newGroupDrawings() =>
    GroupDrawings d = GroupDrawings.new(array.new<line>(), array.new<box>(), array.new<line>(), array.new<label>(),
         newHiddenLabel(label.style_label_down, size.normal), newHiddenLabel(label.style_label_left, size.normal),
         array.new<line>(), array.new<line>(), array.new<linefill>())
    // 先创建云图，再创建K线：后创建的图形显示在上层
    for i = 0 to CLOUD_LINE_COUNT - 1
        line fastLine = newHiddenLine(1)
        line slowLine = newHiddenLine(2)
        d.cloudFastLines.push(fastLine)
        d.cloudSlowLines.push(slowLine)
        d.cloudFills.push(linefill.new(fastLine, slowLine, TRANSPARENT))
    for i = 0 to HTF_BAR_COUNT - 1
        d.wicks.push(newHiddenLine(2))
        d.bodies.push(box.new(bar_index, close, bar_index, close, border_color = TRANSPARENT, bgcolor = TRANSPARENT))
    for i = 0 to 3
        line dashLine = newHiddenLine(1)
        dashLine.set_style(line.style_dashed)
        d.dashLines.push(dashLine)
        d.priceLabels.push(newHiddenLabel(label.style_label_left, size.small))
    d

// 隐藏本组全部图形（每次刷新先全部隐藏，再只显示用到的）
method hideAll(GroupDrawings this) =>
    for ln in this.wicks
        ln.set_color(TRANSPARENT)
    for bx in this.bodies
        bx.set_bgcolor(TRANSPARENT)
        bx.set_border_color(TRANSPARENT)
    for ln in this.dashLines
        ln.set_color(TRANSPARENT)
    for lb in this.priceLabels
        lb.set_text("")
    this.titleLabel.set_text("")
    this.titleLabel.set_color(TRANSPARENT)
    this.warningLabel.set_text("")
    this.warningLabel.set_color(TRANSPARENT)
    for ln in this.cloudFastLines
        ln.set_color(TRANSPARENT)
    for ln in this.cloudSlowLines
        ln.set_color(TRANSPARENT)
    for lf in this.cloudFills
        lf.set_color(TRANSPARENT)
    this

// 绘制函数（全部是更新已有对象）
// ─────────────────────────────────────────────

// 更新一组K线（实体 + 影线），实时K线可附带虚线连接线与价格标签，最后在实时K线上方显示标题标签
// uniformColor：true = 同一涨跌方向的K线颜色完全一致；false = 历史K线偏淡、实时K线更实
drawCandles(GroupDrawings d, array<float> ohlcData, string tfText, int count, array<int> startXs, array<int> endXs, array<int> midXs, bool showLabels, bool showLines, bool showTitleLabel, bool uniformColor) =>
    int   liveMidX = na  // 实时K线中点横坐标，用于定位标题标签
    float liveHigh = na  // 实时K线最高价，用于定位标题标签

    for p = 0 to count - 1
        bool isLive = p == count - 1         // 最后一根为实时K线
        int  base   = (count - 1 - p) * 4    // 该K线在数据数组中的起始下标
        float o = ohlcData.get(base)
        float h = ohlcData.get(base + 1)
        float l = ohlcData.get(base + 2)
        float c = ohlcData.get(base + 3)

        if not na(o) and not na(h) and not na(l) and not na(c)
            int startX = startXs.get(p)
            int endX   = endXs.get(p)
            int midX   = midXs.get(p)

            // 涨跌方向始终各自判断；是否区分实时/历史深浅由 uniformColor 决定
            color baseColor = c >= o ? bullColorInput : bearColorInput
            color wickColor = uniformColor or isLive ? baseColor : color.new(baseColor, HIST_WICK_TRANS)
            int   bodyTrans = uniformColor ? 0 : isLive ? liveBodyTransInput : HIST_BODY_TRANS

            line wick = d.wicks.get(p)
            wick.set_xy1(midX, h)
            wick.set_xy2(midX, l)
            wick.set_color(wickColor)

            box body = d.bodies.get(p)
            body.set_lefttop(startX, math.max(o, c))
            body.set_rightbottom(endX, math.min(o, c))
            body.set_border_color(wickColor)
            body.set_bgcolor(color.new(baseColor, bodyTrans))

            if isLive
                liveMidX := midX
                liveHigh := h

                // 虚线：从图表最新K线连到投影的实时K线开高低收
                if showLines
                    array<float> prices = array.from(o, h, l, c)
                    array<int>   endPts = array.from(startX, midX, midX, endX)
                    array<color> colors = array.from(NEUTRAL_COLOR, bullColorInput, bearColorInput, baseColor)
                    for i = 0 to 3
                        line dashLine = d.dashLines.get(i)
                        dashLine.set_xy1(bar_index, prices.get(i))
                        dashLine.set_xy2(endPts.get(i), prices.get(i))
                        dashLine.set_color(colors.get(i))

                // 价格标签：显示在实时K线右侧
                if showLabels
                    array<float>  prices = array.from(o, h, l, c)
                    array<string> names  = array.from("开盘", "最高", "最低", "收盘")
                    array<color>  colors = array.from(NEUTRAL_COLOR, bullColorInput, bearColorInput, baseColor)
                    for i = 0 to 3
                        label priceLabel = d.priceLabels.get(i)
                        priceLabel.set_xy(endX + 1, prices.get(i))
                        priceLabel.set_text(names.get(i) + ": " + str.tostring(prices.get(i), format.mintick))
                        priceLabel.set_textcolor(colors.get(i))

    // 标题标签（品种代码 + 周期）：贴在实时K线正上方，间距按价格的0.4%计算，避免高周期振幅大时被推出屏幕
    if showTitleLabel and not na(liveMidX)
        float padding = liveHigh > 0 ? liveHigh * 0.004 : syminfo.mintick * 50
        d.titleLabel.set_xy(liveMidX, liveHigh + padding)
        d.titleLabel.set_text(syminfo.ticker + "\n(" + tfText + ")")
        d.titleLabel.set_color(titleLabelBgColorInput)
        d.titleLabel.set_textcolor(titleLabelTextColorInput)
        d.titleLabel.set_style(toLabelStyle(titleLabelStyleInput))
        d.titleLabel.set_size(toLabelSize(titleLabelSizeInput))

// 更新第 cloudNo 条云图（从 0 开始）：相邻两根K线之间各一段快线和慢线，linefill 填充两线之间的区域
// 每段按左端点的快慢关系单独上色，因此金叉/死叉处颜色会自然切换
// showEdgeLine 为 false 时线段保持透明，仅作为 linefill 的锚线
drawCloud(GroupDrawings d, int cloudNo, CloudSpec spec, array<float> fastSlowValues, array<int> midXs, int count, bool showEdgeLine) =>
    color edgeColor = showEdgeLine ? color.white : TRANSPARENT
    if count >= 2  // 至少两根K线才能连成线段（count=1 时 for 0 to -1 会反向循环，必须排除）
        for p = 0 to count - 2
            int   leftOffset  = count - 1 - p           // 左端点距实时K线的偏移
            int   rightOffset = leftOffset - 1          // 右端点距实时K线的偏移
            float fastLeft    = fastSlowValues.get(leftOffset)
            float fastRight   = fastSlowValues.get(rightOffset)
            float slowLeft    = fastSlowValues.get(leftOffset + HTF_BAR_COUNT)
            float slowRight   = fastSlowValues.get(rightOffset + HTF_BAR_COUNT)
            if not na(fastLeft) and not na(fastRight) and not na(slowLeft) and not na(slowRight)
                int  index    = cloudNo * SEGMENT_COUNT + p
                line fastLine = d.cloudFastLines.get(index)
                line slowLine = d.cloudSlowLines.get(index)
                fastLine.set_xy1(midXs.get(p), fastLeft)
                fastLine.set_xy2(midXs.get(p + 1), fastRight)
                fastLine.set_color(edgeColor)
                slowLine.set_xy1(midXs.get(p), slowLeft)
                slowLine.set_xy2(midXs.get(p + 1), slowRight)
                slowLine.set_color(edgeColor)
                d.cloudFills.get(index).set_color(color.new(fastLeft >= slowLeft ? spec.bullColor : spec.bearColor, spec.fillTrans))

// 在分组锚点处显示诊断标签
drawWarning(GroupDrawings d, int groupNo, string message, int x) =>
    d.warningLabel.set_xy(x, close)
    d.warningLabel.set_text("⚠ 周期组" + str.tostring(groupNo) + message)
    d.warningLabel.set_color(color.new(color.red, 10))
    d.warningLabel.set_textcolor(color.white)

// 更新一整组（云图在下层、K线在上层，由对象池的创建顺序保证），取数失败（如周期不受支持）时在锚点处显示诊断标签
drawTimeframeGroup(GroupDrawings d, TfGroup g, int liveAnchor) =>
    [startXs, endXs, midXs] = calcGroupPositions(g.count, liveAnchor, g.width, g.gap)

    for [k, spec] in g.clouds
        if spec.show
            drawCloud(d, k, spec, cloudSlice(g.data, k), midXs, g.count, g.showCloudLine)
