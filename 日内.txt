// ─────────────────────────────────────────────
// 工具函数
// ─────────────────────────────────────────────
toLineStyle(LineStyle style) =>
    switch style
        LineStyle.dotted     => line.style_dotted
        LineStyle.dashed     => line.style_dashed
        LineStyle.arrowLeft  => line.style_arrow_left
        LineStyle.arrowRight => line.style_arrow_right
        LineStyle.arrowBoth  => line.style_arrow_both
        => line.style_solid

toTextAlign(TextAlign align) =>
    switch align
        TextAlign.center => text.align_center
        TextAlign.right  => text.align_right
        => text.align_left

toVerticalAlign(VerticalAlign align) =>
    align == VerticalAlign.bottom ? text.align_bottom : text.align_top

toTextSize(TextSize textSize) =>
    switch textSize
        TextSize.tiny  => size.tiny
        TextSize.small => size.small
        TextSize.large => size.large
        => size.normal

// 价位只在"持平"时才判断穿越，避免新一天价位跳变时误触发警报
crossedUp(level)   => ta.crossover(close, level) and level == level[1]
crossedDown(level) => ta.crossunder(close, level) and level == level[1]

// 数值是否刚发生变化（只在变化时才更新绘图对象）
changed(float x) => not na(x) and (na(x[1]) or x != x[1])

// 数组只保留最近 MAX_TRACKED 个绘图对象，超出时删除最旧的（"显示所有天数"模式下防止无限增长）
pushLine(array<line> list, line item) =>
    array.push(list, item)
    if array.size(list) > MAX_TRACKED
        line.delete(array.shift(list))
pushBox(array<box> list, box item) =>
    array.push(list, item)
    if array.size(list) > MAX_TRACKED
        box.delete(array.shift(list))
pushLabel(array<label> list, label item) =>
    array.push(list, item)
    if array.size(list) > MAX_TRACKED
        label.delete(array.shift(list))

        line.delete(this.ln)

    this.ln  := na
    this.lbl := na
    this

    line.set_y2(this.ln, price)
    label.set_y(this.lbl, price)
    label.set_text(this.lbl, this.textFor(price))
    this

    ahLowLv.freezeOrDelete()
    if timeframe.isintraday
        ahHighLv.draw(ahHigh, ahHighBar)
        ahLowLv.draw(ahLow, ahLowBar)
else if session.ispostmarket
    if high > ahHigh
        ahHigh    := high
        ahHighBar := bar_index
        ahHighLv.moveTo(ahHigh, ahHighBar)
    if low < ahLow
        ahLow    := low
        ahLowBar := bar_index
        ahLowLv.moveTo(ahLow, ahLowBar)

// ─────────────────────────────────────────────
// 前一日 / 前第2日 / CDO / PDCP：每个交易时段首根K线重画
// ─────────────────────────────────────────────
if session.isfirstbar
    prevHighLv.freezeOrDelete()
    prevLowLv.freezeOrDelete()
    prev2HighLv.freezeOrDelete()
    prev2LowLv.freezeOrDelete()
    prevOpenLv.freezeOrDelete()
    cdoLv.freezeOrDelete()
    pdcpLv.freezeOrDelete()
    prevCloseLv.freezeOrDelete()
    if timeframe.isintraday
        prevHighLv.draw(prevHigh, prevHighBar)
        prevLowLv.draw(prevLow, prevLowBar)
        prev2HighLv.draw(prev2High, prev2HighBar)
        prev2LowLv.draw(prev2Low, prev2LowBar)
        prevOpenLv.draw(prevOpen, bar_index)
        cdoLv.draw(cdo, bar_index)
        pdcpLv.draw(pdcp, prevHighBar)
        prevCloseLv.draw(prevClose, bar_index)

// 美股的 CDO 要到常规时段开盘才知道：更新盘前创建的线和标签
if isFirstMarketBar and not session.isfirstbar
    cdoLv.moveTo(cdo, na)

// ─────────────────────────────────────────────
// 日内高低点 / 常规时段开盘价 / 常规时段收盘价
// ─────────────────────────────────────────────
// 日内高低点：新交易时段开始时停用旧线（常规时段开盘时再画新线）
if session.isfirstbar or rthStartNow
    dayHighLv.freezeOrDelete()
    dayLowLv.freezeOrDelete()

if rthStartNow
    rthOpen := open
    rthOpenLv.freezeOrDelete()
    if timeframe.isintraday
        rthOpenLv.draw(rthOpen, bar_index)
        dayHighLv.draw(dayHigh, bar_index)
        dayLowLv.draw(dayLow, bar_index)

// RTH 进行中：日内高低点变化时移动
if isRthActive and not rthStartNow
    if dayHighChanged
        dayHighLv.moveTo(dayHigh, dayHighBar)
    if dayLowChanged
        dayLowLv.moveTo(dayLow, dayLowBar)

// 常规时段收盘价：当天 RTH 结束时定格（没开盘前盘后数据时在下一天常规时段开盘定格），在那之前显示上一个交易日的收盘价
if rthDayCompleted
    rthClose := rthDone1Close
    rthCloseLv.freezeOrDelete()
    if timeframe.isintraday
        rthCloseLv.draw(rthClose, bar_index)

// ─────────────────────────────────────────────
// 上周 / 本周：每周首根K线重画，之后本周高低点变化时移动
// ─────────────────────────────────────────────
if isFirstBarOfWeek
    prevWkHighLv.freezeOrDelete()
    prevWkLowLv.freezeOrDelete()
    wkHighLv.freezeOrDelete()
    wkLowLv.freezeOrDelete()
    if timeframe.isintraday
        prevWkHighLv.draw(prevWkHigh, prevWkHighBar)
        prevWkLowLv.draw(prevWkLow, prevWkLowBar)
        wkHighLv.draw(wkHigh, bar_index)
        wkLowLv.draw(wkLow, bar_index)
else
    // 必须放在重画之后：否则每周首根K线会把上周的旧线挪到本周再定格，画出错位的线
    if wkHighChanged
        wkHighLv.moveTo(wkHigh, wkTrackHighBar)
    if wkLowChanged
        wkLowLv.moveTo(wkLow, wkTrackLowBar)

// 线都是向右无限延伸的，不需要每根K线更新 x2；标签只需在最后一根K线上跟随
if barstate.islast
    prevHighLv.placeLabel()
    prevLowLv.placeLabel()
    prev2HighLv.placeLabel()
    prev2LowLv.placeLabel()
    prevOpenLv.placeLabel()
    cdoLv.placeLabel()
    pdcpLv.placeLabel()
    prevCloseLv.placeLabel()
    ahHighLv.placeLabel()
    ahLowLv.placeLabel()
    dayHighLv.placeLabel()
    dayLowLv.placeLabel()
    prevWkHighLv.placeLabel()
    prevWkLowLv.placeLabel()
    wkHighLv.placeLabel()
    wkLowLv.placeLabel()
    rthOpenLv.placeLabel()
    rthCloseLv.placeLabel()

// ─────────────────────────────────────────────
// ORB 开盘区间（跟随常规时段输入；关掉显示也照样计算，警报可用）
// ─────────────────────────────────────────────
rthEndHour   = int(str.tonumber(str.substring(rthSessionInput, 5, 7)))
rthEndMinute = int(str.tonumber(str.substring(rthSessionInput, 7, 9)))

var int   orbStartTime       = na
var int   orbEndTime         = na
var int   orbCloseTime       = na
var float orbHigh            = na
var float orbLow             = na
var bool  orbLongTriggered   = false
var bool  orbShortTriggered  = false
var array<line>  orbLineList  = array.new<line>()
var array<label> orbLabelList = array.new<label>()

if rthStartNow
    orbStartTime      := time
    orbEndTime        := time + orbDurationInput * 60 * 1000
    orbCloseTime      := timestamp(NY_TZ, year(time, NY_TZ), month(time, NY_TZ), dayofmonth(time, NY_TZ), rthEndHour, rthEndMinute)
    orbHigh           := na
    orbLow            := na
    orbLongTriggered  := false
    orbShortTriggered := false
    // 选"显示所有天数"时保留历史开盘区间，其余选项只显示当天
    if levelDisplayInput != LevelDisplay.allDays
        while array.size(orbLineList) > 0
            line.delete(array.pop(orbLineList))
        while array.size(orbLabelList) > 0
            label.delete(array.pop(orbLabelList))

isInOrb = not na(orbStartTime) and time >= orbStartTime and time < orbEndTime
if isInOrb
    if na(orbHigh) or high > orbHigh
        orbHigh := high
    if na(orbLow) or low < orbLow
        orbLow := low

isOrbEnd = isInOrb[1] and not isInOrb

if showOrbInput and isOrbEnd
    if na(orbHigh) or na(orbLow)
        pushLabel(orbLabelList, label.new(bar_index, high, "开盘区间数据缺失", color = color.red, style = label.style_label_down, textcolor = color.white))
    if not na(orbHigh)
        pushLine(orbLineList, line.new(orbStartTime, orbHigh, orbCloseTime, orbHigh, xloc = xloc.bar_time, color = orbHighColorInput, width = ORB_LINE_WIDTH))
        // 在线的最右端（常规时段收盘时间）显示 ORB 高点价格
        if showPmOrbLabelsInput
            pushLabel(orbLabelList, label.new(orbCloseTime, orbHigh, "ORB 高 (" + str.tostring(orbHigh, format.mintick) + ")", xloc = xloc.bar_time, style = label.style_label_left, color = TRANSPARENT, textcolor = orbHighColorInput, size = pmOrbLabelSize))
    if not na(orbLow)
        pushLine(orbLineList, line.new(orbStartTime, orbLow, orbCloseTime, orbLow, xloc = xloc.bar_time, color = orbLowColorInput, width = ORB_LINE_WIDTH))
        // 在线的最右端（常规时段收盘时间）显示 ORB 低点价格
        if showPmOrbLabelsInput
            pushLabel(orbLabelList, label.new(orbCloseTime, orbLow, "ORB 低 (" + str.tostring(orbLow, format.mintick) + ")", xloc = xloc.bar_time, style = label.style_label_left, color = TRANSPARENT, textcolor = orbLowColorInput, size = pmOrbLabelSize))

// 突破判断：开盘区间结束后、K线收盘确认时判断；跳空直接开在区间外也算突破
if not isInOrb and time >= orbEndTime and not na(orbHigh) and not na(orbLow) and barstate.isconfirmed
    if close > orbHigh and not orbLongTriggered
        orbLongTriggered  := true
        orbShortTriggered := false
        if showOrbInput and showOrbBreakoutLabelsInput
            pushLabel(orbLabelList, label.new(bar_index, orbHigh, "LONG", color = color.blue, textcolor = color.white, style = label.style_flag, size = size.small))
    if close < orbLow and not orbShortTriggered
        orbShortTriggered := true
        orbLongTriggered  := false
        if showOrbInput and showOrbBreakoutLabelsInput
            pushLabel(orbLabelList, label.new(bar_index, orbLow, "SHORT", color = color.yellow, textcolor = color.white, style = label.style_flag, size = size.small))

// 与 LONG / SHORT 标签同一套条件，供警报使用
orbLongSignal  = orbLongTriggered and not orbLongTriggered[1]
orbShortSignal = orbShortTriggered and not orbShortTriggered[1]

// ─────────────────────────────────────────────
// 提示：美股没开盘前盘后数据时，盘前 / 盘后价位不会显示
// ─────────────────────────────────────────────
var bool hasExtendedHours = false
if session.ispremarket or session.ispostmarket
    hasExtendedHours := true
isStockLike = syminfo.type == "stock" or syminfo.type == "fund" or syminfo.type == "dr"
var table ethWarning = table.new(position.bottom_right, 1, 1)
if barstate.islast and timeframe.isintraday and isStockLike and not hasExtendedHours
    table.cell(ethWarning, 0, 0, "未开启盘前盘后数据 (ETH)：盘前 / 盘后价位不会显示", text_color = color.white, bgcolor = color.new(color.red, 20), text_size = size.small)

// ─────────────────────────────────────────────
// 价格轴显示
// ─────────────────────────────────────────────
isAxisEnabled = timeframe.isintraday and isAxisVisibleTf

plot(isAxisEnabled and showPrevHighOnAxisInput ? prevHigh : na, "前一日最高价", color = prevHighColorInput, editable = false, display = display.price_scale)
plot(isAxisEnabled and showPrevLowOnAxisInput ? prevLow : na, "前一日最低价", color = prevLowColorInput, editable = false, display = display.price_scale)
plot(isAxisEnabled and showPrev2HighOnAxisInput ? prev2High : na, "前第2日最高价", color = prev2HighColorInput, editable = false, display = display.price_scale)
plot(isAxisEnabled and showPrev2LowOnAxisInput ? prev2Low : na, "前第2日最低价", color = prev2LowColorInput, editable = false, display = display.price_scale)
plot(isAxisEnabled and showPrevOpenOnAxisInput ? prevOpen : na, "前一日开盘价", color = prevOpenColorInput, editable = false, display = display.price_scale)
plot(isAxisEnabled and showCdoOnAxisInput ? cdo : na, "当日开盘价(CDO)", color = cdoColorInput, editable = false, display = display.price_scale)
plot(isAxisEnabled and showPdcpOnAxisInput ? pdcp : na, "PDCP 前一日控制点", color = pdcpColorInput, editable = false, display = display.price_scale)
plot(isAxisEnabled and showPrevCloseOnAxisInput ? prevClose : na, "前一日收盘价", color = prevCloseColorInput, editable = false, display = display.price_scale)
plot(isAxisEnabled and showAhHighOnAxisInput ? ahHigh : na, "盘后高点", color = ahHighColorInput, editable = false, display = display.price_scale)
plot(isAxisEnabled and showAhLowOnAxisInput ? ahLow : na, "盘后低点", color = ahLowColorInput, editable = false, display = display.price_scale)
plot(isAxisEnabled and showDayHighOnAxisInput ? dayHigh : na, "日内高点", color = dayHighColorInput, editable = false, display = display.price_scale)
plot(isAxisEnabled and showDayLowOnAxisInput ? dayLow : na, "日内低点", color = dayLowColorInput, editable = false, display = display.price_scale)
plot(isAxisEnabled and showPrevWkHighOnAxisInput ? prevWkHigh : na, "上周最高价", color = prevWkHighColorInput, editable = false, display = display.price_scale)
plot(isAxisEnabled and showPrevWkLowOnAxisInput ? prevWkLow : na, "上周最低价", color = prevWkLowColorInput, editable = false, display = display.price_scale)
plot(isAxisEnabled and showWkHighOnAxisInput ? wkHigh : na, "本周最高价", color = wkHighColorInput, editable = false, display = display.price_scale)
plot(isAxisEnabled and showWkLowOnAxisInput ? wkLow : na, "本周最低价", color = wkLowColorInput, editable = false, display = display.price_scale)
plot(isAxisEnabled and showRthOpenOnAxisInput ? rthOpen : na, "常规时段开盘价", color = rthOpenColorInput, editable = false, display = display.price_scale)
plot(isAxisEnabled and showRthCloseOnAxisInput ? rthClose : na, "常规时段收盘价", color = rthCloseColorInput, editable = false, display = display.price_scale)
plot(isAxisEnabled and showOrbInput ? orbHigh : na, "ORB 高点", color = orbHighColorInput, editable = false, display = display.price_scale)
plot(isAxisEnabled and showOrbInput ? orbLow : na, "ORB 低点", color = orbLowColorInput, editable = false, display = display.price_scale)

// ─────────────────────────────────────────────
// 警报（价位跳变的那根K线不触发）
// ─────────────────────────────────────────────
alertcondition(crossedUp(pmHigh), "价格突破盘前高点", "{{ticker}} {{interval}} 价格突破盘前高点，收盘价 {{close}}")
alertcondition(crossedDown(pmLow), "价格跌破盘前低点", "{{ticker}} {{interval}} 价格跌破盘前低点，收盘价 {{close}}")
alertcondition(crossedUp(prevHigh), "价格突破前日高点", "{{ticker}} {{interval}} 价格突破前日高点，收盘价 {{close}}")
alertcondition(crossedDown(prevLow), "价格跌破前日低点", "{{ticker}} {{interval}} 价格跌破前日低点，收盘价 {{close}}")
alertcondition(orbLongSignal, "ORB 向上突破", "{{ticker}} {{interval}} 收盘突破开盘区间高点 (LONG)，收盘价 {{close}}")
alertcondition(orbShortSignal, "ORB 向下突破", "{{ticker}} {{interval}} 收盘跌破开盘区间低点 (SHORT)，收盘价 {{close}}")
