//---------------------
//FUNCTIONS
//---------------------
// 没开盘前盘后数据时，9:30 就是交易时段首根，inRth[1] 不会是 na，所以额外用 session.isfirstbar 判断
isRthStart()        => not na(inRth) and (na(inRth[1]) or session.isfirstbar)
isRthEnd()          => na(inRth) and not na(inRth[1])
isPremarketStart()  => not na(inPremarket) and na(inPremarket[1])
isPremarketEnd()    => na(inPremarket) and not na(inPremarket[1])

// 价位只在"持平"时才判断穿越，避免新一天价位跳变时误触发警报
crossedUp(level)   => ta.crossover(close, level) and level == level[1]
crossedDown(level) => ta.crossunder(close, level) and level == level[1]

// 线型选项 → line.style_*（包括三种箭头）
toLineStyle(opt) =>
    switch opt
        'dotted (┈)'      => line.style_dotted
        'dashed (╌)'      => line.style_dashed
        'arrow left (←)'  => line.style_arrow_left
        'arrow right (→)' => line.style_arrow_right
        'arrows both (↔)' => line.style_arrow_both
        => line.style_solid

toTextAlign(opt) =>
    switch opt
        'align center' => text.align_center
        'align right'  => text.align_right
        => text.align_left

// 数值是否刚发生变化（只在变化时才更新绘图对象）
changed(float x) => not na(x) and (na(x[1]) or x != x[1])

// 数组只保留最近 MAX_TRACKED 个绘图对象，超出时删除最旧的（"显示所有天数"模式下防止无限增长）
MAX_TRACKED = 500
pushLine(array<line> list, line item) =>
    array.push(list, item)
    if array.size(list) > MAX_TRACKED
        line.delete(array.shift(list))
pushBox(array<box> list, box item) =>
    array.push(list, item)
    if array.size(list) > MAX_TRACKED
        box.delete(array.shift(list))
pushLabel(array<label> list, label item) =>
    array.push(list, item)
    if array.size(list) > MAX_TRACKED
        label.delete(array.shift(list))

rthStartNow = isRthStart()
// 结算当天 RTH：开了盘前盘后数据时在 16:00 后第一根；没开时在下一天 9:30 第一根（重置之前）
rthDayCompleted = rthDayOpen and (isRthEnd() or rthStartNow)

if rthDayCompleted
    rthDone2High    := rthDone1High
    rthDone2Low     := rthDone1Low
    rthDone2HighBar := rthDone1HighBar
    rthDone2LowBar  := rthDone1LowBar
    rthDone1High    := rthTrackHigh
    rthDone1Low     := rthTrackLow
    rthDone1HighBar := rthTrackHighBar
    rthDone1LowBar  := rthTrackLowBar
    rthDone1Open    := rthTrackOpen
    rthDone1Close   := rthTrackClose
    rthDayOpen      := false
    isRth           := false

if rthStartNow
    isRth           := true
    rthDayOpen      := true
    rthTrackHigh    := high
    rthTrackLow     := low
    rthTrackHighBar := bar_index
    rthTrackLowBar  := bar_index
    rthTrackOpen    := open

if isRth
    if high > rthTrackHigh
        rthTrackHigh    := high
        rthTrackHighBar := bar_index
    if low < rthTrackLow
        rthTrackLow    := low
        rthTrackLowBar := bar_index
    rthTrackClose := close

newPmLine(float price, color col) =>
    line.new(bar_index, price, bar_index + pmLineExtendBarsInput, price, color = col, width = pmLineWidthInput, style = ulgl.GetLineStyle(pmLineStyleInput))

newPmTextBox(float price, string txt, color col) =>
    x = bar_index + pmLineExtendBarsInput
    box.new(left = x, top = price, right = x, bottom = price, bgcolor = color.new(color.black, 100), border_color = na, text = txt + " (" + str.tostring(price, format.mintick) + ")", text_valign = ulgl.GetVAlign(labelVAlignInput), text_halign = ulgl.GetHAlign(labelHAlignInput), text_size = ulgl.GetFontSize(labelSizeInput), text_color = col)

setPmLevel(line ln, box tb, float price, string txt) =>
    line.set_y1(ln, price)
    line.set_y2(ln, price)
    box.set_top(tb, price)
    box.set_bottom(tb, price)
    box.set_text(tb, txt + " (" + str.tostring(price, format.mintick) + ")")

setPmRight(line ln, box tb, int x) =>
    line.set_x2(ln, x)
    box.set_left(tb, x)
    box.set_right(tb, x)

    if levelDisplay == LevelDisplay.allDays
        // 显示所有天数：前一天的盘前线和标签定格在上一根K线的位置

        setPmRight(pmQ25Line, pmQ25TextBox, prevRightX)

        while array.size(pmBoxList) > 0

    pmLow         := low

        pushLine(pmLineList, pmHighLine)

        pushLine(pmLineList, pmQ75Line)
    if showPmMidLineInput
        pmMidLine := newPmLine(pmMid, pmMidColorInput)
        pushLine(pmLineList, pmMidLine)
    if showPmQ25LineInput
        pmQ25Line := newPmLine(pmQ25, pmQ25ColorInput)
        pushLine(pmLineList, pmQ25Line)
    if showPmLowLineInput
        pmLowLine := newPmLine(pmLow, pmLowColorInput)
        pushLine(pmLineList, pmLowLine)
    if showLabelsInput
        if showPmHighLineInput
            pmHighTextBox := newPmTextBox(pmHigh, "盘前高点", pmHighColorInput)
            pushBox(pmBoxList, pmHighTextBox)
        if showPmQ75LineInput

        if showPmMidLineInput

            pushBox(pmBoxList, pmMidTextBox)

            pushBox(pmBoxList, pmQ25TextBox)
        if showPmLowLineInput

// 盘前高低点变化时才更新价格（没创建的对象是 na，调用不会有任何效果）

    box.set_bottom(pmShadeBox, pmLow)
    setPmLevel(pmHighLine, pmHighTextBox, pmHigh, "盘前高点")
    setPmLevel(pmQ75Line, pmQ75TextBox, pmQ75, "75%")
    setPmLevel(pmMidLine, pmMidTextBox, pmMid, "50%")
    setPmLevel(pmQ25Line, pmQ25TextBox, pmQ25, "25%")
    setPmLevel(pmLowLine, pmLowTextBox, pmLow, "盘前低点")

// 横向位置只需要在最后一根K线上更新（历史上的在下一次盘前开始时定格）

    setPmRight(pmQ75Line, pmQ75TextBox, pmRightX)
    setPmRight(pmMidLine, pmMidTextBox, pmRightX)

//---------------------
//价位对象：每组价位 = 输入设置 + 当前的线和标签
//---------------------
type Level
    bool   showLine
    bool   showLabel
    bool   showPrice
    int    labelOffset
    string labelText
    color  textColor
    string textAlign
    color  lineColor
    int    lineWidth
    string lineStyle
    line   ln  = na
    label  lbl = na

// 画新的线和标签（x1 = 线的起点，na 时从当前K线开始）
method draw(Level this, float price, int x1) =>
    lineExtend = levelDisplay == LevelDisplay.mostRecent ? extend.both : extend.right
    if this.showLine
        this.ln := line.new(na(x1) ? bar_index : x1, price, bar_index + 1, price, color = this.lineColor, style = this.lineStyle, width = this.lineWidth, extend = lineExtend)
    if this.showLabel
        this.lbl := label.new(bar_index + this.labelOffset, price, text = this.showPrice ? this.labelText + str.tostring(price, format.mintick) : this.labelText, color = this.lineColor, style = label.style_label_left, textcolor = this.textColor, size = size.normal, textalign = this.textAlign)
    this

// 停掉旧的线和标签：显示所有天数时定格在上一根K线，否则删除
method retire(Level this) =>
    if levelDisplay == LevelDisplay.allDays
        line.set_x2(this.ln, bar_index - 1)
        line.set_extend(this.ln, extend.none)
        label.set_x(this.lbl, bar_index - 1 + this.labelOffset)
    else
        line.delete(this.ln)
        label.delete(this.lbl)
    this.ln  := na
    this.lbl := na
    this

// 价位变动时移动线和标签（x1 = 新起点，na 表示起点不变）
method moveTo(Level this, float price, int x1) =>
    if not na(x1)
        line.set_x1(this.ln, x1)
    line.set_x2(this.ln, bar_index + 1)
    line.set_y1(this.ln, price)
    line.set_y2(this.ln, price)
    label.set_y(this.lbl, price)
    if this.showPrice
        label.set_text(this.lbl, this.labelText + str.tostring(price, format.mintick))
    this

// 标签跟随最后一根K线
method placeLabel(Level this) =>
    label.set_x(this.lbl, bar_index + this.labelOffset)
    this
